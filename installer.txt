; ============ REGISTRY ENTRIES ============
!macro customInstall
  ; Add registry entries for "Open with Pub POS"
  WriteRegStr HKCU "Software\Pub POS" "InstallPath" "$INSTDIR"
  WriteRegStr HKCU "Software\Pub POS" "Version" "${VERSION}"
  WriteRegStr HKCU "Software\Pub POS" "InstallDate" "$INSTDIR"
  
  ; Add file association for .pubpos files (optional)
  WriteRegStr HKCR ".pubpos" "" "PubPOS.Receipt"
  WriteRegStr HKCR "PubPOS.Receipt" "" "Pub POS Receipt"
  WriteRegStr HKCR "PubPOS.Receipt\DefaultIcon" "" "$INSTDIR\Pub POS.exe,0"
  WriteRegStr HKCR "PubPOS.Receipt\shell\open\command" "" '"$INSTDIR\Pub POS.exe" "%1"'
!macroend

!macro customUnInstall
  ; Clean up registry
  DeleteRegKey HKCU "Software\Pub POS"
  DeleteRegKey HKCR ".pubpos"
  DeleteRegKey HKCR "PubPOS.Receipt"
  
  ; Ask user if they want to delete data
  MessageBox MB_YESNO "Do you want to also delete all Pub POS data (sales, users, licenses)?$\r$\n$\r$\nClick No to keep your data for a future reinstall." IDNO skipDataDelete
    RMDir /r "$APPDATA\pub-pos"
    RMDir /r "$LOCALAPPDATA\pub-pos"
  skipDataDelete:
!macroend
